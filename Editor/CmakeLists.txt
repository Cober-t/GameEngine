# Editor CMakeLists.txt

project(Editor)

# Collect all source files
file(GLOB_RECURSE EDITOR_SOURCES 
    "include/*.h"
    "src/*.cpp"
)

# Add stb_image sources
file(GLOB STB_IMAGE_SOURCES 
    "${CMAKE_SOURCE_DIR}/libraries/stb_image/*.h"
    "${CMAKE_SOURCE_DIR}/libraries/stb_image/*.cpp"
)

# Add GLM sources
file(GLOB_RECURSE GLM_SOURCES 
    "${CMAKE_SOURCE_DIR}/libraries/glm/glm/*.hpp"
    "${CMAKE_SOURCE_DIR}/libraries/glm/glm/*.inl"
)

# Create the editor executable
add_executable(Editor 
    ${EDITOR_SOURCES}
    ${STB_IMAGE_SOURCES}
    ${GLM_SOURCES}
)

# Set subsystem to console
if(WIN32)
    set_target_properties(Editor PROPERTIES
        WIN32_EXECUTABLE FALSE
    )
endif()

# Include directories
target_include_directories(Editor PRIVATE
    include
    include/Panels
    ${CMAKE_SOURCE_DIR}/Engine
    ${CMAKE_SOURCE_DIR}/Engine/include
    ${CMAKE_SOURCE_DIR}/libraries
    ${GLM_INCLUDE_DIR}
    ${ENTT_INCLUDE_DIR}
    ${GLAD_INCLUDE_DIR}
    ${IMGUI_INCLUDE_DIR}
    ${IMGUIZMO_INCLUDE_DIR}
    ${BOX2D_INCLUDE_DIR}
    ${GLFW_INCLUDE_DIR}
    ${SDL3_INCLUDE_DIR}
    ${STB_IMAGE_INCLUDE_DIR}
    ${STB_TRUETYPE_INCLUDE_DIR}
    ${LUA_INCLUDE_DIR}
    ${SOL_INCLUDE_DIR}
    ${SPDLOG_INCLUDE_DIR}
)

# Link libraries - using conditional linking based on target existence
target_link_libraries(Editor PRIVATE
    Engine
    $<$<TARGET_EXISTS:ImGui>:ImGui>
    $<$<TARGET_EXISTS:ImGuizmo>:ImGuizmo>
    $<$<TARGET_EXISTS:glfw>:glfw>
    $<$<TARGET_EXISTS:SDL3-static>:SDL3-static>
    $<$<TARGET_EXISTS:SDL3::SDL3>:SDL3::SDL3>
    $<$<TARGET_EXISTS:Glad>:Glad>
    $<$<TARGET_EXISTS:box2d>:box2d>
)

# Windows-specific libraries
if(WIN32)
    target_link_libraries(Editor PRIVATE opengl32)
endif()

# Editor-specific definitions
target_compile_definitions(Editor PRIVATE
    __EDITOR__
    SOLUTION_DIR="${CMAKE_SOURCE_DIR}/"
    EDITOR_ICONS_DIR="${CMAKE_SOURCE_DIR}/Editor/icons/"
)

# Set output directories
set_target_properties(Editor PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>/Editor
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>/Editor
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin-int/$<CONFIG>/Editor
)

# Set folder in Visual Studio
set_target_properties(Editor PROPERTIES FOLDER "Tools")

# Configuration-specific settings
target_compile_options(Editor PRIVATE
    $<$<CONFIG:Debug>:$<IF:$<CXX_COMPILER_ID:MSVC>,/MDd,>>
    $<$<CONFIG:Release>:$<IF:$<CXX_COMPILER_ID:MSVC>,/MD,>>
    $<$<CONFIG:Dist>:$<IF:$<CXX_COMPILER_ID:MSVC>,/MD,>>
)

# Debug symbols for debug configuration
set_target_properties(Editor PROPERTIES
    $<$<CONFIG:Debug>:CMAKE_CXX_FLAGS_DEBUG>
    $<$<CONFIG:Debug>:"/Zi">
)

# Enable AddressSanitizer for Editor
# enable_address_sanitizer(Editor)

# Make Editor depend on Engine to ensure proper build order
add_dependencies(Editor Engine)
