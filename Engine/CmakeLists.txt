# Engine CMakeLists.txt

project(Engine)

# Collect all source files
file(GLOB_RECURSE ENGINE_SOURCES 
    "src/*.cpp"
    "include/*.h"
)

# Add stb_image sources
file(GLOB STB_IMAGE_SOURCES 
    "${CMAKE_SOURCE_DIR}/libraries/stb_image/*.cpp"
    "${CMAKE_SOURCE_DIR}/libraries/stb_image/*.h"
)

# Add GLM sources
file(GLOB_RECURSE GLM_SOURCES 
    "${CMAKE_SOURCE_DIR}/libraries/glm/glm/*.hpp"
    "${CMAKE_SOURCE_DIR}/libraries/glm/glm/*.inl"
)

# Create the engine library as shared (DLL)
add_library(Engine SHARED 
    ${ENGINE_SOURCES}
    ${STB_IMAGE_SOURCES}
    ${GLM_SOURCES}
)

# Set up precompiled headers
target_precompile_headers(Engine PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_SOURCE_DIR}/include/pch.h>"
)

# Define engine-specific macros
target_compile_definitions(Engine PRIVATE
    _CB_BUILD_DLL
)

# Include directories
target_include_directories(Engine PRIVATE
    include
    include/Core
    include/ImGui
    include/Platforms
    include/Platforms/OpenGL
    include/Render
    include/Render/Text
    include/Render/Primitives
    include/Render/Camera
    include/Scene
    include/Scene/Systems
    ${CMAKE_SOURCE_DIR}/libraries
)

# Public include directories (for consumers of this library)
target_include_directories(Engine PUBLIC
    include
    ${GLFW_INCLUDE_DIR}
    ${SDL3_INCLUDE_DIR}
    ${GLAD_INCLUDE_DIR}
    ${IMGUI_INCLUDE_DIR}
    ${BOX2D_INCLUDE_DIR}
    ${GLM_INCLUDE_DIR}
    ${ENTT_INCLUDE_DIR}
    ${MSDF_INCLUDE_DIR}
    ${MSDFGEN_INCLUDE_DIR}
    ${STB_IMAGE_INCLUDE_DIR}
    ${STB_TRUETYPE_INCLUDE_DIR}
    ${SPDLOG_INCLUDE_DIR}
    ${LUA_INCLUDE_DIR}
    ${SOL_INCLUDE_DIR}
)

# Link libraries - using the actual target names from the submodules
target_link_libraries(Engine PUBLIC
    # Libraries with their own CMake
    $<$<TARGET_EXISTS:glfw>:glfw>
    $<$<TARGET_EXISTS:SDL3-static>:SDL3-static>
    $<$<TARGET_EXISTS:SDL3::SDL3>:SDL3::SDL3>
    $<$<TARGET_EXISTS:box2d>:box2d>
    $<$<TARGET_EXISTS:spdlog>:spdlog>
    $<$<TARGET_EXISTS:glm>:glm>
    $<$<TARGET_EXISTS:msdf-atlas-gen>:msdf-atlas-gen>
    
    # Custom built libraries
    $<$<TARGET_EXISTS:Glad>:Glad>
    $<$<TARGET_EXISTS:ImGui>:ImGui>
    $<$<TARGET_EXISTS:Lua>:Lua>
    
    # Header-only libraries
    $<$<TARGET_EXISTS:entt>:entt>
    $<$<TARGET_EXISTS:sol>:sol>
    $<$<TARGET_EXISTS:stb_image>:stb_image>
    $<$<TARGET_EXISTS:stb_truetype>:stb_truetype>
)

# Windows-specific libraries
if(WIN32)
    target_link_libraries(Engine PUBLIC opengl32)
endif()

# Set output directories
set_target_properties(Engine PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin-int/$<CONFIG>/Engine
)

# Post-build commands to copy DLL to Editor and Game directories
if(WIN32)
    add_custom_command(TARGET Engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:Engine>
        ${CMAKE_BINARY_DIR}/bin/$<CONFIG>/Editor/
        COMMENT "Copying Engine DLL to Editor directory"
    )
    
    add_custom_command(TARGET Engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory
        ${CMAKE_SOURCE_DIR}/Game/build/$<CONFIG>
        COMMENT "Creating Game build directory"
    )
    
    add_custom_command(TARGET Engine POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:Engine>
        ${CMAKE_SOURCE_DIR}/Game/build/$<CONFIG>/
        COMMENT "Copying Engine DLL to Game directory"
    )
endif()

# Set folder in Visual Studio
set_target_properties(Engine PROPERTIES FOLDER "Core")

# Configuration-specific settings
target_compile_options(Engine PRIVATE
    $<$<CONFIG:Debug>:$<IF:$<CXX_COMPILER_ID:MSVC>,/Od /ZI,>>
    $<$<CONFIG:Release>:$<IF:$<CXX_COMPILER_ID:MSVC>,/O2,>>
    $<$<CONFIG:Dist>:$<IF:$<CXX_COMPILER_ID:MSVC>,/O2,>>
)

# Debug symbols
set_target_properties(Engine PROPERTIES
    $<$<CONFIG:Debug>:CMAKE_CXX_FLAGS_DEBUG>
    $<$<CONFIG:Debug>:"/Zi">
)
