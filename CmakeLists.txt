# AddressSanitizer configuration
function(enable_address_sanitizer target)
    if(ENABLE_ASAN AND CMAKE_BUILD_TYPE STREQUAL "Debug")
        message(STATUS "AddressSanitizer enabled for target: ${target}")
        
        if(MSVC)
            # MSVC AddressSanitizer flags
            target_compile_options(${target} PRIVATE 
                $<$<CONFIG:Debug>:/fsanitize=address>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:/fsanitize=address>
            )
            
            # Link with AddressSanitizer runtime
            target_link_options(${target} PRIVATE 
                $<$<CONFIG:Debug>:/INCREMENTAL:NO>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:/INCREMENTAL:NO>
            )
            
            # Set environment variable for better stack traces
            target_compile_definitions(${target} PRIVATE
                $<$<CONFIG:Debug>:_DISABLE_VECTOR_ANNOTATION>
                $<$<CONFIG:Debug>:_DISABLE_STRING_ANNOTATION>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:_DISABLE_VECTOR_ANNOTATION>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:_DISABLE_STRING_ANNOTATION>
            )
            
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            # GCC/Clang AddressSanitizer flags
            target_compile_options(${target} PRIVATE 
                $<$<CONFIG:Debug>:-fsanitize=address>
                $<$<CONFIG:Debug>:-fno-omit-frame-pointer>
                $<$<CONFIG:Debug>:-g>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:-fsanitize=address>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:-fno-omit-frame-pointer>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:-g>
            )
            
            target_link_options(${target} PRIVATE 
                $<$<CONFIG:Debug>:-fsanitize=address>
                $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:-fsanitize=address>
            )
        endif()
        
        # AddressSanitizer-specific definitions
        target_compile_definitions(${target} PRIVATE
            $<$<CONFIG:Debug>:ASAN_ENABLED>
            $<$<AND:$<BOOL:${ENABLE_ASAN}>,$<NOT:$<CONFIG:Debug>>>:ASAN_ENABLED>
        )
    endif()
endfunction()
